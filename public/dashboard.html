Classe Completa C++ (libcurl + nlohmann/json)

#pragma once
#define CURL_STATICLIB
#include <string>
#include <curl/curl.h>
#include <nlohmann/json.hpp>
#include <windows.h>
#include <Wbemidl.h>
#include <comdef.h>

#pragma comment(lib, "libcurl_a.lib")
#pragma comment(lib, "ws2_32.lib")
#pragma comment(lib, "wldap32.lib")
#pragma comment(lib, "crypt32.lib")
#pragma comment(lib, "normaliz.lib")
#pragma comment(lib, "wbemuuid.lib")

using json = nlohmann::json;

static size_t WriteCallback(void* contents, size_t size, size_t nmemb, void* userp) {
    ((std::string*)userp)->append((char*)contents, size * nmemb);
    return size * nmemb;
}

struct AuthResult {
    bool success = false;
    std::string message;
    std::string session_id;
    std::string username;
    std::string expiry;
    int level = 0;
};

class KeyAuthClient {
private:
    std::string baseUrl;
    std::string appId;
    std::string appSecret;
    std::string sessionId;

public:
    KeyAuthClient(const std::string& url,
                  const std::string& id,
                  const std::string& secret)
        : baseUrl(url), appId(id), appSecret(secret) {}

    // POST request generico
    json Post(const std::string& endpoint, const json& body) {
        CURL* curl = curl_easy_init();
        if (!curl) return json{{"success", false}, {"message", "Erro CURL"}};

        std::string response;
        std::string url = baseUrl + endpoint;
        std::string data = body.dump();

        struct curl_slist* headers = nullptr;
        headers = curl_slist_append(headers, "Content-Type: application/json");

        curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
        curl_easy_setopt(curl, CURLOPT_POSTFIELDS, data.c_str());
        curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, &response);
        curl_easy_setopt(curl, CURLOPT_TIMEOUT, 10L);
        curl_easy_setopt(curl, CURLOPT_SSL_VERIFYPEER, 0L);

        CURLcode res = curl_easy_perform(curl);
        curl_slist_free_all(headers);
        curl_easy_cleanup(curl);

        if (res != CURLE_OK)
            return json{{"success", false}, {"message", "Erro de conexao"}};

        return json::parse(response, nullptr, false);
    }

    // 1. Inicializar app
    bool Init() {
        json body;
        body["app_id"]     = appId;
        body["app_secret"] = appSecret;
        body["version"]    = "1.0";

        json res = Post("/init", body);
        if (res.value("success", false)) {
            sessionId = res.value("session_id", "");
            return true;
        }
        return false;
    }

    // 2. Login
    AuthResult Login(const std::string& user,
                     const std::string& pass,
                     const std::string& hwid) {
        AuthResult result;
        json body;
        body["app_id"]   = appId;
        body["username"] = user;
        body["password"] = pass;
        body["hwid"]     = hwid;

        json res = Post("/login", body);
        result.success = res.value("success", false);
        if (result.success) {
            result.session_id = res.value("session_id", "");
            result.username   = res.value("username", "");
            result.level      = res.value("level", 1);
            result.expiry     = res.value("expiry", "");
            result.message    = "Login OK";
        } else {
            result.message = res.value("message", "Erro");
        }
        return result;
    }

    // 3. Registro
    AuthResult Register(const std::string& user,
                        const std::string& pass,
                        const std::string& key,
                        const std::string& email,
                        const std::string& hwid) {
        AuthResult result;
        json body;
        body["app_id"]   = appId;
        body["username"] = user;
        body["password"] = pass;
        body["key"]      = key;
        body["email"]    = email;
        body["hwid"]     = hwid;

        json res = Post("/register", body);
        result.success = res.value("success", false);
        result.message = res.value("message", "");
        return result;
    }

    // 4. Ativar licenca
    AuthResult License(const std::string& key,
                       const std::string& hwid) {
        AuthResult result;
        json body;
        body["app_id"] = appId;
        body["key"]    = key;
        body["hwid"]   = hwid;

        json res = Post("/license", body);
        result.success = res.value("success", false);
        if (result.success) {
            result.level  = res.value("level", 1);
            result.expiry = res.value("expiry", "");
        }
        result.message = res.value("message", "");
        return result;
    }

    // 5. Verificar sessao
    bool CheckSession(const std::string& sid) {
        json body;
        body["session_id"] = sid;
        json res = Post("/check-session", body);
        return res.value("success", false);
    }

    // 6. Obter variavel
    std::string GetVar(const std::string& varName) {
        json body;
        body["app_id"]   = appId;
        body["var_name"] = varName;
        json res = Post("/var", body);
        if (res.value("success", false))
            return res.value("value", "");
        return "";
    }

    const std::string& GetSessionId() const { return sessionId; }
};

Exemplo de uso no main.cpp

#include "KeyAuthClient.h"

int main() {
    curl_global_init(CURL_GLOBAL_ALL);

    KeyAuthClient auth(
        "http://localhost:3000/api/client",
        "SEU-APP-ID",
        "SEU-APP-SECRET"
    );

    // Inicializar
    if (!auth.Init()) {
        printf("Erro ao inicializar\n");
        return 1;
    }

    // Login
    std::string hwid = GenerateHWID();
    AuthResult login = auth.Login("usuario", "senha", hwid);

    if (login.success) {
        printf("Bem-vindo %s!\n", login.username.c_str());
        printf("Nivel: %d\n", login.level);
        printf("Expira: %s\n", login.expiry.c_str());
        printf("Session: %s\n", login.session_id.c_str());

        // Obter variavel
        std::string val = auth.GetVar("minha_var");
        printf("Variavel: %s\n", val.c_str());

        // Verificar sessao
        if (auth.CheckSession(login.session_id))
            printf("Sessao valida!\n");
    } else {
        printf("Erro: %s\n", login.message.c_str());
    }

    curl_global_cleanup();
    return 0;
}

// Gerar HWID (Windows)
std::string GenerateHWID() {
    HKEY hKey;
    if (RegOpenKeyExA(HKEY_LOCAL_MACHINE,
        "SOFTWARE\\Microsoft\\Cryptography",
        0, KEY_READ | KEY_WOW64_64KEY, &hKey) != ERROR_SUCCESS)
        return "UNKNOWN";

    char buffer[256];
    DWORD size = sizeof(buffer);
    RegQueryValueExA(hKey, "MachineGuid", NULL, NULL,
                     (LPBYTE)buffer, &size);
    RegCloseKey(hKey);
    return std::string(buffer);
}
